"); }); const otpStore = {}; const BAD_WORDS_REGEX = /(كلب|حمار|منيوك|قحبة|شرموط|عرص|طيز|كس|زب|يلعن|قواد|زق|سافل|حقير)/i; app.post('/api/send-otp', (req, res) => { const { email } = req.body; if (!email) return res.status(400).json({ error: 'حط إيميلك يا الأمير!' }); const code = Math.floor(1000 + Math.random() * 9000).toString(); otpStore[email] = { code, expires: Date.now() + 5 * 60 * 1000 }; console.log(`🔑 كود الدخول للإيميل [${email}]: ${code}`); res.json({ message: 'تم إرسال الرمز بنجاح!', debugCode: code }); const rooms = {}; io.on('connection', (socket) => { socket.on('create_room', ({ username, email, isOwner, avatar, gameType }) => { const roomCode = Math.random().toString(36).substring(2, 6).toUpperCase(); rooms[roomCode] = { code: roomCode, hostId: socket.id, gameType: gameType || 'BLUFF', settings: { writeTime: 30, guessTime: 20, maxRounds: 5 }, round: 0, state: 'LOBBY', players: [{ id: socket.id, username, email, isOwner, avatar, score: 0 }], currentQ: null, submittedAnswers: {}, votes: {} }; socket.join(roomCode); socket.emit('room_created', rooms[roomCode]); }); socket.on('join_room', ({ roomCode, username, email, isOwner, avatar }) => { const room = rooms[roomCode?.toUpperCase()]; if (!room) return socket.emit('error_msg', 'الكود غلط يا الحبيب ما فيه غرفة كذا!'); if (room.state !== 'LOBBY') return socket.emit('error_msg', 'اللعب بادي، انتظر للجولة الجاية!'); room.players.push({ id: socket.id, username, email, isOwner, avatar, score: 0 }); socket.join(room.code); io.to(room.code).emit('room_updated', room); }); socket.on('start_bluff_round', (roomCode) => { const room = rooms[roomCode]; if (!room) return; if (room.players.length < 3) { return socket.emit('error_msg', 'الماب ما يكفي! لازم 3 متكين على الأقل في جنتل بلاي!'); } room.state = 'BLUFF_WRITING'; room.submittedAnswers = {}; room.votes = {}; room.round++; const qIndex = Math.floor(Math.random() * BLUFF_QUESTIONS.length); room.currentQ = BLUFF_QUESTIONS[qIndex]; io.to(roomCode).emit('bluff_question_phase', { question: room.currentQ.q, category: room.currentQ.category, timeLimit: room.settings.writeTime, round: room.round }); }); socket.on('submit_fake_answer', ({ roomCode, fakeAnswer }) => { const room = rooms[roomCode]; if (!room) return; room.submittedAnswers[socket.id] = fakeAnswer.trim(); if (Object.keys(room.submittedAnswers).length === room.players.length) { room.state = 'BLUFF_GUESSING'; const choices = [{ text: room.currentQ.a, authorId: 'REAL' }]; for (const pid in room.submittedAnswers) { choices.push({ text: room.submittedAnswers[pid], authorId: pid }); } choices.sort(() => Math.random() - 0.5); room.currentChoices = choices; io.to(room.code).emit('bluff_guessing_phase', { question: room.currentQ.q, choices: choices.map(c => c.text), timeLimit: room.settings.guessTime }); } }); socket.on('submit_vote', ({ roomCode, chosenText }) => { const room = rooms[roomCode]; if (!room) return; room.votes[socket.id] = chosenText; if (Object.keys(room.votes).length === room.players.length) { const realAns = room.currentQ.a; const scoresGained = {}; room.players.forEach(p => scoresGained[p.id] = 0); for (const voterId in room.votes) { const vote = room.votes[voterId]; if (vote === realAns) { scoresGained[voterId] += 500; } else { for (const [authorId, fakeText] of Object.entries(room.submittedAnswers)) { if (fakeText === vote && authorId !== voterId) { scoresGained[authorId] += 300; } } } } room.players.forEach(p => p.score += (scoresGained[p.id] || 0)); io.to(room.code).emit('bluff_round_results', { realAnswer: realAns, players: room.players, scoresGained }); } }); socket.on('disconnect', () => { for (const code in rooms) { const room = rooms[code]; room.players = room.players.filter(p => p.id !== socket.id); if (room.players.length === 0) delete rooms[code]; else io.to(code).emit('room_updated', room); } }); }); const PORT = process.env.PORT || 3000;
